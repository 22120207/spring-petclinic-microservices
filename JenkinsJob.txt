pipeline {
    agent any

    parameters {
        string(name: 'CUSTOMERS_IMAGE_TAG', defaultValue: 'latest', description: 'Tag for customers image')
        string(name: 'VETS_IMAGE_TAG',      defaultValue: 'latest', description: 'Tag for vets image')
        string(name: 'VISITS_IMAGE_TAG',    defaultValue: 'latest', description: 'Tag for visits image')
        string(name: 'GENAI_IMAGE_TAG',     defaultValue: 'latest', description: 'Tag for genai image')
    }

    environment {
        USERNAME = "tienminhktvn2"
    }

    stages {
        stage('Deploy Images') {
            steps {
                script {
                    def modules = [
                        [name: 'spring-petclinic-customers-service', tag: params.CUSTOMERS_IMAGE_TAG],
                        [name: 'spring-petclinic-vets-service',      tag: params.VETS_IMAGE_TAG],
                        [name: 'spring-petclinic-visits-service',    tag: params.VISITS_IMAGE_TAG],
                        [name: 'spring-petclinic-gen-ai',            tag: params.GENAI_IMAGE_TAG]
                    ]
                    modules.each { m ->
                        echo "${USERNAME}/${m.name}:${m.tag}"
                    }
                }
            }
        }
    }
}
